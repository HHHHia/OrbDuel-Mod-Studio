cmake_minimum_required(VERSION 3.20)
project(my_mod LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Visual Studio 2022 x64 for OrbDuel mods")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/mod.json" MANIFEST)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/mod.json")
function(setting output fallback)
  string(JSON value ERROR_VARIABLE error GET "${MANIFEST}" ${ARGN})
  if(error)
    set(value "${fallback}")
  endif()
  set(${output} "${value}" PARENT_SCOPE)
endfunction()
setting(MOD_ID "my-mod" id)
setting(LIBRARY "my_mod.dll" library)
setting(TYPE_SUFFIX "vortex" typeId)
setting(TYPE_NAME "旋风" typeName)
setting(STRENGTH "300" defaults strength)
string(JSON strengthType ERROR_VARIABLE strengthError TYPE "${MANIFEST}" defaults strength)
if(NOT strengthError AND NOT strengthType STREQUAL "NUMBER")
  message(FATAL_ERROR "defaults.strength must be a number")
endif()
if(NOT MOD_ID MATCHES "^[a-z][a-z0-9_-]*$" OR NOT TYPE_SUFFIX MATCHES "^[a-z][a-z0-9_-]*$")
  message(FATAL_ERROR "id/typeId must start with a lowercase letter and use letters, numbers, - or _")
endif()
if(NOT LIBRARY MATCHES "^[A-Za-z0-9_.-]+[.]dll$")
  message(FATAL_ERROR "library must be a DLL filename without a directory")
endif()
if(STRENGTH MATCHES "^-?[0-9]+$")
  string(APPEND STRENGTH ".0")
endif()
set(TYPE_ID "${MOD_ID}.${TYPE_SUFFIX}")
foreach(var TYPE_NAME TYPE_ID)
  string(REPLACE "\\" "\\\\" ${var} "${${var}}")
  string(REPLACE "\"" "\\\"" ${var} "${${var}}")
  string(REPLACE "\n" "\\n" ${var} "${${var}}")
  string(REPLACE "\r" "\\r" ${var} "${${var}}")
endforeach()
setting(interval "2" defaults interval)
if(interval MATCHES "^-?[0-9]+$")
  string(APPEND interval ".0")
endif()
setting(maxAlive "2" defaults maxAlive)
if(maxAlive MATCHES "^-?[0-9]+$")
  string(APPEND maxAlive ".0")
endif()
setting(lifetime "8" defaults lifetime)
if(lifetime MATCHES "^-?[0-9]+$")
  string(APPEND lifetime ".0")
endif()
setting(hpScale "0.35" defaults hpScale)
if(hpScale MATCHES "^-?[0-9]+$")
  string(APPEND hpScale ".0")
endif()
setting(radiusScale "0.65" defaults radiusScale)
if(radiusScale MATCHES "^-?[0-9]+$")
  string(APPEND radiusScale ".0")
endif()
setting(attackScale "0.5" defaults attackScale)
if(attackScale MATCHES "^-?[0-9]+$")
  string(APPEND attackScale ".0")
endif()
setting(speed "280" defaults speed)
if(speed MATCHES "^-?[0-9]+$")
  string(APPEND speed ".0")
endif()
configure_file(mod_config.hpp.in generated/mod_config.hpp @ONLY)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/include/orbduel_mod.hpp")
  set(SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/include")
else()
  set(SDK_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
endif()
if(NOT EXISTS "${SDK_INCLUDE}/orbduel_mod.hpp")
  message(FATAL_ERROR "SDK header missing: create a project with mod-sdk/new-mod.ps1 or copy include/ into it")
endif()
add_library(my_mod SHARED my_mod.cpp)
target_include_directories(my_mod PRIVATE "${SDK_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(my_mod PRIVATE /utf-8)
target_compile_definitions(my_mod PRIVATE ORBDUEL_USER_MOD)
string(REGEX REPLACE "[.]dll$" "" OUTPUT_NAME "${LIBRARY}")
set(PACKAGE "${CMAKE_BINARY_DIR}/package/${MOD_ID}")
file(MAKE_DIRECTORY "${PACKAGE}")
configure_file(mod.json "${PACKAGE}/mod.json" COPYONLY)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/SDK_LICENSE.txt")
  configure_file(SDK_LICENSE.txt "${PACKAGE}/SDK_LICENSE.txt" COPYONLY)
else()
  configure_file("${CMAKE_CURRENT_SOURCE_DIR}/../LICENSE.txt" "${PACKAGE}/SDK_LICENSE.txt" COPYONLY)
endif()
if(IS_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/assets" DESTINATION "${PACKAGE}")
endif()
set_target_properties(my_mod PROPERTIES OUTPUT_NAME "${OUTPUT_NAME}" RUNTIME_OUTPUT_DIRECTORY "${PACKAGE}")
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(my_mod PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${cfg} "${PACKAGE}")
endforeach()
