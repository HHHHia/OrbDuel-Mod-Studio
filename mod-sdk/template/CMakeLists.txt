cmake_minimum_required(VERSION 3.20)
project(my_mod LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(my_mod SHARED my_mod.cpp)
target_include_directories(my_mod PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../include")
if(MSVC)
  target_compile_options(my_mod PRIVATE /utf-8)
endif()
set_target_properties(my_mod PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/my-mod")
foreach(cfg DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(my_mod PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${cfg} "${CMAKE_BINARY_DIR}/package/my-mod")
endforeach()
add_custom_command(TARGET my_mod POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CMAKE_CURRENT_SOURCE_DIR}/mod.json" "$<TARGET_FILE_DIR:my_mod>/mod.json")
