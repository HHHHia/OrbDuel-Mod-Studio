cmake_minimum_required(VERSION 3.20)
project(orbduel VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "OrbDuel requires a 64-bit toolchain")
endif()
function(orbduel_cxx target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W3 /utf-8)
  endif()
endfunction()
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Windows builds use MSVC 2022 x64")
  endif()
  set(SDL2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/sdl2/msvc")
  set(SDL2_INCLUDE_DIR "${SDL2_ROOT}/include")
  set(SDL2_LINK_LIBS "${SDL2_ROOT}/lib/SDL2main.lib" "${SDL2_ROOT}/lib/SDL2.lib" "${SDL2_ROOT}/lib/SDL2_image.lib" "${SDL2_ROOT}/lib/SDL2_ttf.lib")
elseif(APPLE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
  pkg_check_modules(SDL2_IMAGE REQUIRED IMPORTED_TARGET SDL2_image)
  pkg_check_modules(SDL2_TTF REQUIRED IMPORTED_TARGET SDL2_ttf)
  set(SDL2_INCLUDE_DIR ${SDL2_INCLUDE_DIRS})
  set(SDL2_LINK_LIBS PkgConfig::SDL2 PkgConfig::SDL2_IMAGE PkgConfig::SDL2_TTF)
else()
  message(FATAL_ERROR "Supported targets: Windows x64 and macOS (Apple Clang)")
endif()
add_library(orbduel_logic STATIC
  src/core/Event.cpp src/core/Log.cpp src/core/Rng.cpp src/core/Time.cpp src/core/version.cpp
  src/physics/Arena.cpp src/physics/Collision.cpp src/physics/World.cpp
  src/content/ContentLoader.cpp src/content/Scene.cpp src/mods/PluginManager.cpp
  src/match/Match.cpp src/match/MatchRule.cpp src/match/Skill.cpp src/match/StallGuard.cpp src/match/Zone.cpp src/match/Simulation.cpp)
target_include_directories(orbduel_logic PUBLIC include third_party mod-sdk/include)
orbduel_cxx(orbduel_logic)
add_library(imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
  third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
  third_party/imgui/backends/imgui_impl_sdl2.cpp third_party/imgui/backends/imgui_impl_sdlrenderer2.cpp)
target_include_directories(imgui PUBLIC third_party/imgui PRIVATE ${SDL2_INCLUDE_DIR})
add_library(orbduel_engine STATIC src/compile_check.cpp src/render/AssetCache.cpp
  src/render/Renderer.cpp src/render/Themes.cpp src/game/Audio.cpp src/game/Studio.cpp)
target_include_directories(orbduel_engine PUBLIC include PRIVATE ${SDL2_INCLUDE_DIR})
target_link_libraries(orbduel_engine PUBLIC orbduel_logic PRIVATE imgui ${SDL2_LINK_LIBS})
orbduel_cxx(orbduel_engine)
add_executable(orbduel src/game/main.cpp)
target_include_directories(orbduel PRIVATE ${SDL2_INCLUDE_DIR})
target_link_libraries(orbduel PRIVATE orbduel_engine ${SDL2_LINK_LIBS})
orbduel_cxx(orbduel)
foreach(cfg "" _DEBUG _RELEASE _RELWITHDEBINFO _MINSIZEREL)
  set_target_properties(orbduel PROPERTIES RUNTIME_OUTPUT_DIRECTORY${cfg} "${CMAKE_BINARY_DIR}/bin")
endforeach()
file(GLOB SDL2_DLLS "${SDL2_ROOT}/bin/*.dll")
foreach(dll ${SDL2_DLLS})
  add_custom_command(TARGET orbduel POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${dll}" "$<TARGET_FILE_DIR:orbduel>")
endforeach()
if(WIN32)
  set(ORBDUEL_NATIVE_SUFFIX ".dll")
else()
  set(ORBDUEL_NATIVE_SUFFIX ".dylib")
endif()
foreach(mod portal gravity launch lava summon rewind guardian)
  add_library(${mod} SHARED "mod-sdk/examples/${mod}.cpp")
  target_include_directories(${mod} PRIVATE mod-sdk/include)
  orbduel_cxx(${mod})
  foreach(cfg "" _DEBUG _RELEASE _RELWITHDEBINFO _MINSIZEREL)
    set_target_properties(${mod} PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY${cfg} "${CMAKE_BINARY_DIR}/bin/mods/${mod}" LIBRARY_OUTPUT_DIRECTORY${cfg} "${CMAKE_BINARY_DIR}/bin/mods/${mod}")
  endforeach()
  configure_file("mods/${mod}/mod.json" "generated/${mod}/source.json" COPYONLY)
  file(READ "${CMAKE_BINARY_DIR}/generated/${mod}/source.json" mod_manifest)
  string(JSON mod_manifest SET "${mod_manifest}" library "\"${mod}${ORBDUEL_NATIVE_SUFFIX}\"")
  file(WRITE "${CMAKE_BINARY_DIR}/generated/${mod}/mod.json" "${mod_manifest}\n")
  add_custom_command(TARGET ${mod} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${CMAKE_BINARY_DIR}/generated/${mod}/mod.json" "$<TARGET_FILE_DIR:${mod}>/mod.json")
  add_dependencies(orbduel ${mod})
endforeach()
